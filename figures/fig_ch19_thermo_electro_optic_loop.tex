% Three-domain coupling with all feedback routed outside the cards.
\begin{figure}[htbp]
  \centering
  \begin{tikzpicture}[x=1cm,y=1cm,>=Latex]
    \tikzset{
      domain/.style={draw=#1!70!black,very thick,rounded corners=5pt,fill=#1!10,
        minimum width=3.45cm,minimum height=1.55cm,align=center},
      interface/.style={font=\footnotesize,fill=white,inner sep=2.5pt,align=center}
    }

    \node[domain=blue] (elec) at (-4.2,1.55)
      {电学与注入\\$J,\ n,\ p,\ \mu,\rho_c$};
    \node[domain=red] (therm) at (0,1.55)
      {热传导\\$\nabla\!\cdot(k\nabla T)+Q_{\rm heat}=0$};
    \node[domain=green] (optic) at (4.2,1.55)
      {光学与增益\\$n(T),\ g(N,T),\ \alpha(T)$};

    \draw[->,very thick,orange!85!black] (elec.east)--(therm.west)
      node[interface,midway,above] {焦耳热、复合热\\$Q_{\rm heat}$};
    \draw[->,very thick,red!75!black] (therm.east)--(optic.west)
      node[interface,midway,above] {温度场回写\\$\Delta n,\Delta g,\Delta\alpha$};

    % Outer return path: no arrow crosses any card.
    \draw[->,very thick,blue!70!black,rounded corners=8pt]
      (optic.south) -- (4.2,-0.45) -- (-4.2,-0.45) -- (elec.south)
      node[interface,pos=0.48,below] {受激复合与吸收改变载流子、功率和热源};

    \node[draw=gray!55,rounded corners,fill=gray!5,align=center,font=\footnotesize,
      minimum width=11.5cm,minimum height=0.85cm] at (0,-1.65)
      {时间尺度分离：光学 $\tau_o\sim\mathrm{ps}$\quad
       电学 $\tau_e\sim\mathrm{ns}$\quad
       热学 $\tau_{\rm th}\sim\mu\mathrm{s}$--$\mathrm{ms}$};

    \node[font=\footnotesize,align=center,text width=12cm] at (0,-2.65)
      {求解目标不是依次“跑完三个模块”，而是寻找三者接口量同时收敛的工作点。};
  \end{tikzpicture}
  \caption{PCSEL 的热--电--光闭环。电学模块向热学模块提供焦耳热和复合热；
  温度场通过折射率、增益、吸收和迁移率回写光学与电学参数；
  受激复合与吸收又改变载流子分布、输出功率和热源。箭头均表示模块间传递的接口量，
  不表示三个物理过程具有相同时间尺度。}
  \label{fig:ch19_thermo_electro_optic_loop}
