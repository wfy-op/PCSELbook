  \node[wave] (rx) at (-5.0,2.0) {$R_x$};
  \node[wave] (sx) at (-2.2,2.0) {$S_x$};

  \draw[<->,thick,red!70!black] (ry) -- node[below,fill=white,inner sep=1pt] {$\kappa_1$} (sy);
  \draw[<->,thick,orange!85!black] (rx) -- node[left=5pt] {$\kappa_2$} (ry);

    \begin{bmatrix}R_x\\S_x\\R_y\\S_y\end{bmatrix}$};
  \node[shortnote,anchor=north] at (3.6,-.05) {

    \textcolor{orange!85!black}{$\kappa_2$} 连接 $x/y$ 向基底，表示二维转角散射；虚线箭头是对开放端口的泄漏。线性阈值由矩阵最大实部过零确定，而不是对某一条闭合箭头单独做“环路增益”判断。
